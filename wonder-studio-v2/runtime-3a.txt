function buildDust() {
  const count = 580, p = new Float32Array(count * 3);
  for (let i = 0; i < count; i++) { p[i * 3] = (Math.random() - .5) * 25; p[i * 3 + 1] = Math.random() * 9.4 + .2; p[i * 3 + 2] = (Math.random() - .5) * 19; }
  const g = new THREE.BufferGeometry(); g.setAttribute('position', new THREE.BufferAttribute(p, 3));
  dust.points = new THREE.Points(g, new THREE.PointsMaterial({ color: 0xffe8bb, size: .025, transparent: true, opacity: .32, depthWrite: false })); scene.add(dust.points);
