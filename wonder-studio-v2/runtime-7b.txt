function initialize() {
  try { renderer = new THREE.WebGLRenderer({ antialias: true, powerPreference: 'high-performance', preserveDrawingBuffer: true }); }
  catch (_) { ui.loading.classList.add('hidden'); ui.error.classList.add('show'); return; }
  renderer.setPixelRatio(Math.min(devicePixelRatio, 1.7)); renderer.setSize(innerWidth, innerHeight); renderer.shadowMap.enabled = true; renderer.outputColorSpace = THREE.SRGBColorSpace; renderer.toneMapping = THREE.ACESFilmicToneMapping; document.body.prepend(renderer.domElement);
  scene = new THREE.Scene(); camera = new THREE.PerspectiveCamera(48, innerWidth / innerHeight, .08, 100); camera.position.set(12, 8.5, 16);
  controls = new OrbitControls(camera, renderer.domElement); controls.target.set(0, 3, -1.3); controls.enableDamping = true; controls.minDistance = 5.4; controls.maxDistance = 26; controls.maxPolarAngle = Math.PI * .49; controls.minPolarAngle = .25; controls.enablePan = true;
  raycaster = new THREE.Raycaster(); pointer = new THREE.Vector2(); clock = new THREE.Clock(); dragPlane = new THREE.Plane(); world = new RAPIER.World({ x: 0, y: -9.81, z: 0 }); world.timestep = 1 / 60;
  selectionBox = new THREE.BoxHelper(new THREE.Object3D(), 0xffd98a); selectionBox.material.transparent = true; selectionBox.material.opacity = .8; selectionBox.visible = false; scene.add(selectionBox);
  buildRoom(); buildCatalog();
  if (!restore()) STARTING_COMPONENTS.forEach(([key, p, r]) => createComponent(key, p, r));
  bindUI(); pushHistory(false); animate();
  document.body.dataset.engineReady = 'true';
  window.dispatchEvent(new CustomEvent('wonder-engine-ready'));
  setTimeout(() => { ui.loading.classList.add('hidden'); say('Loose materials now fall, collide, balance, and carry force through real connections.', 5200); }, 900);
}

function animate() {
  requestAnimationFrame(animate); const dt = Math.min(clock.getDelta(), .05), elapsed = clock.elapsedTime; accumulator += dt;
  while (accumulator >= 1 / 60) { world.step(); evaluateInvention(1 / 60, elapsed); accumulator -= 1 / 60; }
  syncPhysics(); controls.update();
  if (!reducedMotion) { dust.points.rotation.y += dt * .008; dust.points.position.y = Math.sin(elapsed * .08) * .18; if (lanternAwake) room.wind.rotation.z += dt * .9; mechanism.crank.rotation.z += (0 - mechanism.crank.rotation.z) * dt * .35; }
  if (lanternAwake) room.lanternLight.intensity = 6.4 + Math.sin(elapsed * 2.2) * .6;
  renderer.render(scene, camera);
}
